◆ pico_input_event_timeout()

int pico_input_event_timeout ( Pico_Event * evt,
int type,
int timeout )

Stops the program until a matching event occurs or a timeout is reached. All input functions delegate to this one. Internal events (quit/exit, window resize, ctrl+zoom/scroll/grid) are handled automatically and never forwarded. On timeout, evt->type is set to PICO_EVENT_NONE.

// file: event_timeout.c
#include <pico.h>
int main() {
pico_open(NULL);
Pico_Event event;
printf("Waiting for an event for the next 2 seconds...\n");
pico_input_event_timeout(&event, 0, 5000);
if (event.type != PICO_EVENT_NONE) {
printf("Event detected! Type: %d\n", event.type);
} else {
printf("No event detected in those 2 seconds\n");
}
pico_close();
return 0;
}
